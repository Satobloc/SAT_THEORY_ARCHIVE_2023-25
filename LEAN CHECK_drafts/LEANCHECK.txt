section Geometry_Proofs

variables (M : Type) [manifold M] [tangent_bundle M] [metric g : tensor_type 0 2 M]

-- Riemann curvature tensor
def Riemann : tensor_type 1 3 M := riemann_tensor ∇ g

-- Ricci tensor
def Ricci : tensor_type 0 2 M := ricci_tensor g ∇

-- Scalar curvature
def R_scalar : M → ℝ := scalar_curvature g ∇

-- First Bianchi identity
theorem bianchi_first (μ ν ρ σ τ : ℕ) :
  ∇_μ (Riemann μ ν ρ σ) + ∇_ν (Riemann ν ρ μ σ) + ∇_ρ (Riemann ρ μ ν σ) = 0 :=
begin
  exact riemann_bianchi_identity ∇,
end

end Geometry_Proofs

section SATO_Mass_Hierarchy

-- Topological invariant Q

-- Topological mass suppression
def m_topological (m_0 Q : ℝ) : ℝ := m_0 / Q

begin
