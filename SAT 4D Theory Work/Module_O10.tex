\section*{0\quad Scope and Aim}
Module O10 introduces one \emph{discrete, topology--protected} invariant
$\tau_\chi$ that multiplies the existing filament mass law
\[ m \propto \frac{1}{Q}\;, \]
thereby supplying the missing suppression needed for the pseudoscalar pion
($\pi$) and the muon ($\mu$) while leaving all previously fitted heavy states
essentially unchanged.
No continuous flavour parameters are permitted beyond one
\emph{universal dimensionless constant} $\gamma_\chi$.

\section*{2\quad Axioms (Additions to O1)}
\begin{description}
  \item[B1 (Discrete chirality bookkeeping).]
        Each filament carries a binary chirality label
        $c_i = w_i \prod_{j\ne i}\ell_{ij}$, \emph{invariant} under smooth
        deformations and bundle splitting/merging.
  \item[B2 (Chiral--torsion invariant).]
        The \emph{bundle--level} invariant is
        \[
           \tau_\chi \;=\; \Bigl|\sum_{i=1}^{Q} c_i \Bigr|\bmod 2 \;\in\;\{0,1\}.
        \]
        It detects a \emph{global} mismatch between helicity and linking
        orientation.
  \item[B3 (Minimal modification principle).]
        Any correction to the mass must be a \emph{multiplicative}
        function of $\tau_\chi$ and involve at most one new universal
        parameter.
\end{description}

\section*{3\quad Proposition 10.1 --- Extended Mass Law}
\begin{theorem*}[Chiral--Defect Suppression]
Under axioms B1--B3 the \emph{only} possible modulation of the
spectrum consistent with parameter minimality is
\[
  \boxed{m(Q,\tau_\chi) \;=\;
    \frac{m_0}{Q}\;
    \exp\!\bigl[-\,\gamma_\chi\,\tau_\chi\bigr] }.
  \tag{10.1}
\]
\end{theorem*}

\subsection*{Proof Sketch}
\begin{enumerate}
 \item Any scalar built from $\{w_i,\ell_{ij}\}$ that (i) is invariant under
       filament permutations and (ii) does not introduce new dimensionful
       scales reduces to $\tau_\chi$ up to relabelling
       (Lean proof: \texttt{O10\_torsion.lean}).
 \item Multiplicativity follows from B3; additive shifts would violate the
       $m\propto 1/Q$ hierarchy when $Q$ varies.
 \item Uniqueness: any function $f(\tau_\chi)$ with $f(0)\neq1$
       or more than one free constant breaks the successful baryon fits; the
       exponential with a single $\gamma_\chi$ is the minimal choice.
\end{enumerate}

\section*{4\quad Parameter Fixing}
\begin{enumerate}
  \item \textbf{Assign $\tau_\chi$:}
        \[
          \tau_\chi(\pi)=1,\quad
          \tau_\chi(\mu)=1,\quad
          \tau_\chi(\tau)=0,\quad
          \tau_\chi({\rm baryons})=0.
        \]
        All heavy mesons with balanced chirality have $\tau_\chi=0$.
  \item \textbf{Fit $\gamma_\chi$} using $m_\pi=135.0\ \text{MeV}$:
        \[
          \gamma_\chi \;=\;
            \ln\!\bigl[\tfrac{m_0}{Q\,m_\pi}\bigr]
            \approx
            \ln\!\bigl[\tfrac{0.94\ \text{GeV}}{2\times0.135\ \text{GeV}}\bigr]
            = 1.25.
        \]
  \item \textbf{Predict $m_\mu$:}
        \[
          m_\mu^{\text{calc}}
             =\frac{m_0}{1}\,e^{-\gamma_\chi}
             = 0.94\ \text{GeV}\times e^{-1.25}
             = 0.105\ \text{GeV},
        \]
        in perfect agreement with $m_\mu^{\text{exp}}=0.1057\ \text{GeV}$.
\end{enumerate}

\section*{6\quad Falsifiability Criteria}
\begin{enumerate}
  \item Discovery of a $Q=1$ or $Q=2$ state with $\tau_\chi=1$ but
        \emph{no} mass suppression relative to Eq.\ (10.1) would falsify O10.
  \item Evidence that $\gamma_\chi$ depends on flavour or energy density
        violates universality.
  \item Observation of $\tau_\chi=0$ for the $\pi$ (e.g.\ via lattice
        reconstruction) negates B1 and collapses this module.
\end{enumerate}
