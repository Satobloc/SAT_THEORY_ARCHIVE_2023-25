\subsection{I.2~ Metric Emergence Theorem}
Let $F(x)\subset\mathcal F$ denote filaments intersecting $x$.  
Define the ensemble co--metric
\[
  \tilde g_{\mu\nu}(x)=\bigl\langle v_\mu v_\nu\bigr\rangle_{F(x)},\qquad 
  v^\mu=\dot\gamma^\mu.
\]
\paragraph{Theorem} $\tilde g_{\mu\nu}(x)$ is invertible and Lorentzian (\,$-,+,+,+$\,) iff
\begin{enumerate}
\item \emph{Density}: $\operatorname{rank}\tilde g_{\mu\nu}=4$ (filaments span $T_x^\ast M$).
\item \emph{Non--degeneracy}: $\det\tilde g_{\mu\nu}\neq0$.
\item \emph{Anisotropy}: there exists a dominant eigenvector $u^\mu$ with
      $\tilde g_{\mu\nu}u^\mu u^\nu<0$.
\end{enumerate}
Failure of any condition yields Euclidean signature, degenerate metric or absence of emergent time.
