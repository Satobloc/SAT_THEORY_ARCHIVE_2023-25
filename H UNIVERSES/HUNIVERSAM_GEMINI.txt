# 2. PARSER
# ==========================================
def parse_universes(data_string):
    """Splits the raw text into distinct universe dictionary objects."""
    blocks = re.split(r'(?=\d{9}\sH{1,2})', data_string.strip())
    universes = []
    
    for block in blocks:
        if not block.strip(): continue
        
        # Extract ID and Type
        header_match = re.search(r'(\d{9})\s(HH|H)', block)
        if not header_match: continue
        uid, utype = header_match.groups()
        
        # Extract Kinematics (h* and ω*)
        h_rates = [float(val) for val in re.findall(r'@([+-]\d\.\d+)h\*', block)]
        w_rates = [float(val) for val in re.findall(r'@([+-]\d\.\d+)ω\*', block)]
        
        # Extract Coupling Moduli (g)
        g_vals = [float(val) for val in re.findall(r'\[g=([+-]\d\.\d+)\]', block)]
        if not g_vals: g_vals = [0.0] # Default to uncoupled if no g is present
        
        universes.append({
            'id': uid,
            'type': utype,
            'h_rates': h_rates,
            'w_rates': w_rates,
            'g_vals': g_vals,
            'raw': block
        })
    return universes

def calculate_metrics(univ):
    """Applies the conceptual 4D physics formulas to the parsed kinematics."""
    h = np.array(univ['h_rates']) if univ['h_rates'] else np.array([0.0])
    w = np.array(univ['w_rates']) if univ['w_rates'] else np.array([0.0])
    g = np.array(univ['g_vals'])
    
    avg_g = np.mean(g)
    abs_g = np.mean(np.abs(g))
    
    # 1. Hookean Metric Stress: σ = h + Σ(g * kinematic_forces)
    # A proxy using the Euclidean norm of linear strain compounded by torsion tension
    base_strain = np.linalg.norm(h)
    torsion_tension = np.sum(np.abs(w)) * abs_g
    total_stress = base_strain + torsion_tension
    
    # 2. Frustration Index (Directional Mass potential)
    # U ∝ |g| * h^2 * ω^2. Evaluates the max clash of flow and spin.
    max_h = np.max(np.abs(h)) if h.size > 0 else 0
    max_w = np.max(np.abs(w)) if w.size > 0 else 0
    frustration = abs_g * (max_h**2) * (max_w**2)
    
    # 3. Helicity Score (Screw-Thread Index)
    # H = Σ g * (h * ω). Indicates if the universe acts as a pump.
    # Using averages as a macroscopic measure of net helicity
    helicity = avg_g * np.mean(h) * np.mean(w) if h.size > 0 and w.size > 0 else 0
    
    # 4. Gearing Ratio (Net Torsion Flux)
    # Checks for rotational runaway vs rotational deadlock
    w_cw = w[w > 0]
    w_ccw = w[w < 0]
    cw_flux = np.sum(w_cw) if w_cw.size > 0 else 0
    ccw_flux = np.sum(np.abs(w_ccw)) if w_ccw.size > 0 else 0
    
    # Net flux normalized by coupling grip
    gearing_ratio = (cw_flux - ccw_flux) * avg_g
    
    return {
        'stress': total_stress,
        'frustration': frustration,
        'helicity': helicity,
        'gearing': gearing_ratio,
        'cw_flux': cw_flux,
        'ccw_flux': ccw_flux
    }

How the Calculation Engine Works
Parsing Phase: It scans the block using regular expressions, picking up any scalar followed by h*, ω*, or enclosed in [g=...]. It separates the data so the arrays of dimensional flow and angular torsion can be evaluated independently.

Hookean Metric Stress: The engine calculates the Euclidean norm of all linear expansion/contraction (h 
∗
 ). It then adds the sum of all torsional rates (∣ω 
∗
 ∣) scaled by the average absolute coupling modulus (∣g∣). This provides a single scalar output indicating how badly the geometry is being warped.

Frustration Index: It squares the maximum linear and rotational rates, multiplying them by the average absolute coupling strength. Squaring them eliminates negative vectors, focusing purely on the raw magnitude of energy a stationary particle must absorb to remain immobile.

Helicity Score & Gearing Ratio: By combining the directional vectors (keeping the signs of + clockwise and - widdershins), the engine evaluates if the active dimensions form a synchronized mechanism (like a screw-thread flow) or if the gears are deadlocked and grinding.

import os

import random

# ==========================================
# 1. FILE DISCOVERY & FILTERING
# ==========================================
def find_and_read_random_file(directory=".", max_size_mb=50):
    """
    Scans the target directory for .txt files starting with specific strings,
    filters out files larger than max_size_mb, and returns the contents of one at random.
    """
    valid_prefixes = ("SIXTY THOUSAND UNIVERSES", "RANDOM_H_UNIVERSES")
    max_size_bytes = max_size_mb * 1024 * 1024
    valid_files = []

    for filename in os.listdir(directory):
        if not filename.lower().endswith(".txt"):
            continue
            
        # Check if the file starts with any of the requested strings
        # (Replacing underscores with spaces to catch slight naming variations)
        normalized_name = filename.replace("_", " ").upper()
        normalized_prefixes = [p.replace("_", " ").upper() for p in valid_prefixes]
        
        if any(normalized_name.startswith(p) for p in normalized_prefixes):
            filepath = os.path.join(directory, filename)
            
            # Filter out ridiculously large files
            if os.path.getsize(filepath) <= max_size_bytes:
                valid_files.append(filepath)
                
    if not valid_files:
        print(f"No valid .txt files under {max_size_mb}MB matching the prefixes were found in '{directory}'.")
        return None, None

    # Pick a file at random
    selected_file = random.choice(valid_files)
    print(f"Selected Data File: {selected_file} ({os.path.getsize(selected_file) / (1024*1024):.2f} MB)")
    
    # Read and return its contents
    with open(selected_file, 'r', encoding='utf-8', errors='ignore') as f:
        return selected_file, f.read()

def find_and_read_random_file(directory=".", max_size_mb=50):
    valid_prefixes = ("SIXTY THOUSAND UNIVERSES", "RANDOM_H_UNIVERSES")
    max_size_bytes = max_size_mb * 1024 * 1024
    valid_files = []

    for filename in os.listdir(directory):
        if not filename.lower().endswith(".txt"):
            continue
            
        normalized_name = filename.replace("_", " ").upper()
        normalized_prefixes = [p.replace("_", " ").upper() for p in valid_prefixes]
        
        if any(normalized_name.startswith(p) for p in normalized_prefixes):
            filepath = os.path.join(directory, filename)
            if os.path.getsize(filepath) <= max_size_bytes:
                valid_files.append(filepath)
                
    if not valid_files:
        print(f"No valid .txt files under {max_size_mb}MB found.")

def parse_discrete_universes(data_string):
    """Parses line-by-line discrete states instead of continuous rates."""
    universes = []
    lines = data_string.strip().split('\n')
    
    for line in lines:
        if not line.strip() or not line[0].isdigit():
            continue
            
        # Extract ID and Morphology Bracket
        match = re.match(r'^(\d{9})\s+(\[.*?\])\s+(.*)', line)
        if not match: continue
        
        uid, morphology, kinematics = match.groups()
        
        # Helper to extract dimensions (e.g., "x+y+w" -> ['x', 'y', 'w'])
        def extract_dims(keyword):
            k_match = re.search(fr'{keyword}:\s*([a-z+]+)', kinematics)
            return k_match.group(1).split('+') if k_match else []
            
        universes.append({
            'id': uid,
            'morphology': morphology,
            'exp': extract_dims('expansion'),
            'con': extract_dims('contraction'),
            'cw': extract_dims('clockwise rotation'),
            'ccw': extract_dims('widdershins rotation'),
            'raw': line
        })
    return universes

    strain_factor = 0.6 # How dramatic the stretch/squash is
    
    if 'x' in univ['exp']: sx += strain_factor
    if 'x' in univ['con']: sx -= strain_factor

    if 'y' in univ['con']: sy -= strain_factor

    if 'z' in univ['con']: sz -= strain_factor
