This file implements a conservative mathematical scaffold:

1. Universal Indicatrix:
      Y(s) = r(s) R(s) x0,

No mass formula, particle identification, preferred constants, lattice,

ClearAll["Global`*"];

(* ================================================================ *)
(* 1. so(4) BASIS AND GENERATORS                                    *)
(* ================================================================ *)

planeOrder = {
  {1, 2}, {1, 3}, {1, 4},
  {2, 3}, {2, 4}, {3, 4}
};

so4Basis[i_Integer, j_Integer] /; i != j :=
  Normal @ SparseArray[
    {
      {i, j} -> 1,
      {j, i} -> -1
    },
    {4, 4}
  ];

so4FromVector[v_List] /; Length[v] == 6 :=
  Total @ MapThread[
    #1 so4Basis[#2[[1]], #2[[2]]] &,
    {v, planeOrder}
  ];

lieInner[a_?MatrixQ, b_?MatrixQ] :=
  -Tr[a . b]/2;

skewResidual[m_?MatrixQ] :=
  N @ Norm[m + Transpose[m], "Frobenius"];

(* ================================================================ *)
(* 2. Spin(4) = SU(2)+ x SU(2)- DECOMPOSITION                       *)
(* ================================================================ *)

(*
Let
  B_i = {J23, J31, J12}
and
  A_i = {J14, J24, J34}.

Then the self-dual and anti-self-dual subspaces are spanned by
  T_i(+) = (B_i + A_i)/2
  T_i(-) = (B_i - A_i)/2.

The coefficient extraction below uses the Gram matrix, so it does
not depend on an assumed normalization convention.
*)

spatialDualBasis = {
  so4Basis[2, 3],
  so4Basis[3, 1],
  so4Basis[1, 2]
};

radialPlaneBasis = {
  so4Basis[1, 4],
  so4Basis[2, 4],
  so4Basis[3, 4]
};

plusBasis =
  MapThread[(#1 + #2)/2 &, {spatialDualBasis, radialPlaneBasis}];

minusBasis =
  MapThread[(#1 - #2)/2 &, {spatialDualBasis, radialPlaneBasis}];

coordinatesInBasis[m_?MatrixQ, basis_List] :=
  Module[{gram, rhs},
    gram = Table[
      lieInner[basis[[i]], basis[[j]]],
      {i, Length[basis]},
      {j, Length[basis]}
    ];
    rhs = Table[
      lieInner[basis[[i]], m],
      {i, Length[basis]}
    ];
    LinearSolve[gram, rhs]
  ];

spin4Coordinates[m_?MatrixQ] :=
  Module[{coords},
    coords = coordinatesInBasis[m, Join[plusBasis, minusBasis]];
    <|
      "Plus" -> Take[coords, 3],
      "Minus" -> Drop[coords, 3]
    |>
  ];
