class ConflictResolutionTests(unittest.TestCase):
    def test_newer_candidate_supersedes_old_fact(self) -> None:
        current = AssertionSnapshot(
            assertion_id="old",
            conflict_key="k",
            observed_at=datetime(2026, 4, 1, tzinfo=UTC),
            confidence=0.80,
            normalized_value="boston",
            qualifiers={},
        )
        candidate = AssertionSnapshot(
            assertion_id="new",
            conflict_key="k",
            observed_at=datetime(2026, 4, 10, tzinfo=UTC),
            confidence=0.82,
            normalized_value="san francisco",
            qualifiers={},
        )
        decision = resolve_conflict(current, candidate)
        self.assertEqual(decision.status, "current")
        self.assertTrue(decision.supersede_current)

    def test_change_signal_overrides_current(self) -> None:
        current = AssertionSnapshot(
            assertion_id="old",
            conflict_key="k",
            observed_at=datetime(2026, 4, 10, tzinfo=UTC),
            confidence=0.95,
            normalized_value="new york",
            qualifiers={},
        )
        candidate = AssertionSnapshot(
            assertion_id="new",
            conflict_key="k",
            observed_at=datetime(2026, 4, 11, tzinfo=UTC),
            confidence=0.70,
            normalized_value="london",
            qualifiers={"change_signal": True},
        )
        decision = resolve_conflict(current, candidate)
        self.assertEqual(decision.status, "current")
        self.assertTrue(decision.supersede_current)

    def test_conflicting_values_are_preserved(self) -> None:
        current = AssertionSnapshot(
            assertion_id="old",
            conflict_key="k",
            observed_at=datetime(2026, 4, 10, tzinfo=UTC),
            confidence=0.81,
            normalized_value="tea",
            qualifiers={},
        )
        candidate = AssertionSnapshot(
            assertion_id="new",
            conflict_key="k",
            observed_at=datetime(2026, 4, 10, tzinfo=UTC),
            confidence=0.82,
            normalized_value="coffee",
            qualifiers={},
        )
        decision = resolve_conflict(current, candidate)
        self.assertEqual(decision.status, "conflicting")

    def test_newer_candidate_with_low_confidence_does_not_win(self) -> None:
        current = AssertionSnapshot(
            assertion_id="old",
            conflict_key="k",
            observed_at=datetime(2026, 4, 1, tzinfo=UTC),

        self.assertEqual(decision.status, "historical")

def _write_json(path: Path, payload: Any) -> None:
    path.write_text(json.dumps(payload, ensure_ascii=True, indent=2, sort_keys=True, default=str) + "\n", encoding="utf-8")


def _write_readonly_tree(root: Path) -> None:
    for dirpath, dirnames, filenames in os.walk(root):
        os.chmod(dirpath, 0o555)
        for dirname in dirnames:
            os.chmod(Path(dirpath) / dirname, 0o555)
        for filename in filenames:
            os.chmod(Path(dirpath) / filename, 0o444)


@dataclass(slots=True)
class FilesystemProjection:
    store: Any

    def materialize(self, *, root_dir: str | Path, container_tag: str = "default") -> dict[str, int]:
        root = Path(root_dir).expanduser().resolve()
        scoped_root = root / container_tag
        if scoped_root.exists():
            shutil.rmtree(scoped_root)
        (scoped_root / "current_facts").mkdir(parents=True, exist_ok=True)
        (scoped_root / "assertions").mkdir(parents=True, exist_ok=True)
        (scoped_root / "observations").mkdir(parents=True, exist_ok=True)
        (scoped_root / "episodes").mkdir(parents=True, exist_ok=True)
        (scoped_root / "runtime_events").mkdir(parents=True, exist_ok=True)
        (scoped_root / "reminders").mkdir(parents=True, exist_ok=True)
        (scoped_root / "wiki").mkdir(parents=True, exist_ok=True)

        facts = self.store.list_current_facts(container_tag=container_tag, limit=500)
        assertions = self.store.list_assertions(container_tag=container_tag, limit=500)
        observations = self.store.list_recent_observations(container_tag=container_tag, limit=500)
        episodes = self.store.list_recent_episodes(container_tag=container_tag, limit=200)
        runtime_events = self.store.list_runtime_events(container_tag=container_tag, limit=500)
        reminders = self.store.list_reminders(container_tag=container_tag, limit=128)
        wiki_rows = self.store._fetchall(
            """
            SELECT path, title, page_type, body_md, outbound_links_json, updated_at
            FROM wiki_pages
            WHERE path LIKE %s
            ORDER BY path ASC
            """,
            (f"{container_tag}--%",),
        )

        for row in facts:
            name = f"{row['predicate']}--{slugify(row.get('subject_name') or row['subject_entity_id'])}--{slugify(row.get('object_name') or row.get('literal_value_text') or row['current_fact_id']) or row['current_fact_id']}.json"
            _write_json(scoped_root / "current_facts" / name, row)
        for row in assertions:
            name = f"{row['assertion_id']}--{row['predicate']}.json"
            _write_json(scoped_root / "assertions" / name, row)
        for row in observations:
            name = f"{row['observation_id']}--{row['observation_type']}.json"
            _write_json(scoped_root / "observations" / name, row)
        for row in episodes:
            name = f"{row['episode_id']}--{slugify(row['title'])}.json"
            _write_json(scoped_root / "episodes" / name, row)
        for row in runtime_events:
            name = f"{row['event_id']}--{row['event_type']}.json"
            _write_json(scoped_root / "runtime_events" / name, row)
        for row in reminders:
            name = f"{row['reminder_id']}--{row['scope_kind']}--{slugify(row['scope_id'])}.json"
            _write_json(scoped_root / "reminders" / name, row)
        for row in wiki_rows:
            relative = str(row["path"]).removeprefix(f"{container_tag}--")
            path = scoped_root / "wiki" / f"{relative}.md"
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_text(str(row["body_md"]), encoding="utf-8")

        _write_readonly_tree(scoped_root)
        return {
            "current_facts": len(facts),
            "assertions": len(assertions),
            "observations": len(observations),
            "episodes": len(episodes),
            "runtime_events": len(runtime_events),
            "reminders": len(reminders),
            "wiki_pages": len(wiki_rows),
        }

@dataclass(frozen=True, slots=True)
class ClaimPolicy:
    predicate: str
    resolution_policy: str
    modality: str = "actual"
    polarity: str = "+"
    importance: float = 0.5
    half_life_days: int = 365
    type_weight: float = 1.0


_SINGLETON_POLICIES = {
    "lives_in": ClaimPolicy("lives_in", "latest_wins", importance=0.9, half_life_days=365 * 5, type_weight=1.3),
    "works_at": ClaimPolicy("works_at", "latest_wins", importance=0.9, half_life_days=365 * 5, type_weight=1.3),
    "birthday": ClaimPolicy("birthday", "latest_wins", importance=0.9, half_life_days=365 * 5, type_weight=1.3),
    "allergic_to": ClaimPolicy("allergic_to", "latest_wins", importance=0.9, half_life_days=365 * 5, type_weight=1.3),
    "favorite": ClaimPolicy("favorite", "latest_wins", importance=0.85, half_life_days=365 * 2, type_weight=1.3),
    "prefers": ClaimPolicy("prefers", "latest_wins", importance=0.8, half_life_days=365, type_weight=1.3),
    "has_pet_name": ClaimPolicy("has_pet_name", "latest_wins", importance=0.8, half_life_days=365 * 5, type_weight=1.3),
    "latest_metric_value": ClaimPolicy("latest_metric_value", "latest_wins", importance=0.7, half_life_days=30, type_weight=1.3),
    "activity_duration": ClaimPolicy("activity_duration", "latest_wins", importance=0.8, half_life_days=365, type_weight=1.3),
    "current_record": ClaimPolicy("current_record", "latest_wins", importance=0.78, half_life_days=90, type_weight=1.3),
}

_MERGE_POLICIES = {
    "interest_topic": ClaimPolicy("interest_topic", "merge", importance=0.6, half_life_days=365, type_weight=1.0),
    "interest_domain": ClaimPolicy("interest_domain", "merge", importance=0.6, half_life_days=365, type_weight=1.0),
    "bookmarked_url": ClaimPolicy("bookmarked_url", "merge", importance=0.58, half_life_days=365, type_weight=0.8),
    "attended_event": ClaimPolicy("attended_event", "merge", importance=0.7, half_life_days=365 * 2, type_weight=1.0),
    "started_media": ClaimPolicy("started_media", "merge", importance=0.72, half_life_days=365 * 2, type_weight=1.0),
    "successful_item": ClaimPolicy("successful_item", "merge", importance=0.82, half_life_days=365, type_weight=1.0),
}

_DEFAULT_POLICY = ClaimPolicy("default", "keep_all", importance=0.5, half_life_days=180, type_weight=0.8)


def policy_for_predicate(predicate: str | None) -> ClaimPolicy:
    normalized = (predicate or "").strip()
    if normalized in _SINGLETON_POLICIES:
        return _SINGLETON_POLICIES[normalized]
    if normalized in _MERGE_POLICIES:
        return _MERGE_POLICIES[normalized]
    return ClaimPolicy(
        predicate=normalized or _DEFAULT_POLICY.predicate,
        resolution_policy=_DEFAULT_POLICY.resolution_policy,
        modality=_DEFAULT_POLICY.modality,
        polarity=_DEFAULT_POLICY.polarity,
        importance=_DEFAULT_POLICY.importance,
        half_life_days=_DEFAULT_POLICY.half_life_days,
        type_weight=_DEFAULT_POLICY.type_weight,
    )


def type_weight_for_row(row: dict[str, Any]) -> float:
    predicate = str(row.get("predicate") or "")
    observation_type = str(row.get("observation_type") or "")
    if observation_type:
        return 0.8
    return policy_for_predicate(predicate).type_weight


def half_life_days_for_row(row: dict[str, Any]) -> int:
    predicate = str(row.get("predicate") or "")
    if predicate:
        return policy_for_predicate(predicate).half_life_days
    observation_type = str(row.get("observation_type") or "")
    if observation_type == "health_metric":
        return 30
    if observation_type in {"dialog_turn", "dialog_session"}:
        return 180
    return _DEFAULT_POLICY.half_life_days


def importance_for_predicate(predicate: str | None) -> float:
    return policy_for_predicate(predicate).importance


def resolution_policy_for_predicate(predicate: str | None) -> str:
    return policy_for_predicate(predicate).resolution_policy


def singleton_predicates() -> set[str]:
    return set(_SINGLETON_POLICIES.keys())

def _load_dotenv_if_present() -> None:
    env_path = Path.cwd() / ".env"
    if not env_path.exists():
        return
    for line in env_path.read_text(encoding="utf-8").splitlines():
        stripped = line.strip()
        if not stripped or stripped.startswith("#") or "=" not in stripped:
            continue
        key, value = stripped.split("=", 1)
        os.environ.setdefault(key.strip(), _normalize_env_value(value))


def _normalize_env_value(value: str) -> str:
    normalized = value.strip()
    if len(normalized) >= 2 and normalized[0] == normalized[-1] and normalized[0] in {'"', "'"}:
        return normalized[1:-1]
    return normalized


@dataclass(slots=True)
class EmbeddingConfig:
    api_key: str | None
    base_url: str
    model: str
    enabled: bool
    timeout_seconds: int = 30

    @classmethod
    def from_env(cls) -> "EmbeddingConfig":
        _load_dotenv_if_present()
        api_key = os.environ.get("MEMO_REE_EMBED_API_KEY") or os.environ.get("OPENAI_API_KEY") or None
        mode = (os.environ.get("MEMO_REE_EMBED_MODE") or "enabled").strip().casefold()
        return cls(
            api_key=api_key,
            base_url=(os.environ.get("MEMO_REE_EMBED_BASE_URL") or "https://api.openai.com/v1").rstrip("/"),
            model=os.environ.get("MEMO_REE_EMBED_MODEL") or "text-embedding-3-small",
            enabled=mode not in {"off", "disabled", "false", "0"} and bool(api_key),
            timeout_seconds=int(os.environ.get("MEMO_REE_EMBED_TIMEOUT_SECONDS", "30")),
        )


class TextEmbedder:
    def __init__(self, config: EmbeddingConfig) -> None:
        self.config = config

    @classmethod
    def from_env(cls) -> "TextEmbedder":
        return cls(EmbeddingConfig.from_env())

    def embed_one(self, text: str) -> list[float] | None:
        vectors = self.embed_texts([text])
        if not vectors:
            return None
        return vectors[0]

    def embed_texts(self, texts: list[str]) -> list[list[float]]:
        if not self.config.enabled:
            return []
        clean_texts = [text.strip() for text in texts if text and text.strip()]
        if not clean_texts:
            return []
        body = {
            "model": self.config.model,
            "input": clean_texts,
        }
        req = request.Request(
            f"{self.config.base_url}/embeddings",
            data=json.dumps(body, ensure_ascii=True).encode("utf-8"),
            headers={
                "Content-Type": "application/json",
                "Authorization": f"Bearer {self.config.api_key}",
            },
            method="POST",
        )
        try:
            with request.urlopen(req, timeout=self.config.timeout_seconds) as response:
                payload = json.loads(response.read().decode("utf-8"))
        except error.HTTPError as exc:
            raise RuntimeError(exc.read().decode("utf-8", errors="ignore")) from exc
        data = payload.get("data")
        if not isinstance(data, list):
            raise RuntimeError("Embedding API returned invalid payload")
        vectors: list[list[float]] = []
        for item in data:
            embedding = item.get("embedding") if isinstance(item, dict) else None
            if not isinstance(embedding, list):
                continue
            vectors.append([float(value) for value in embedding])
        return vectors

    @staticmethod
    def vector_literal(vector: list[float]) -> str:
        return json.dumps(vector, ensure_ascii=True, separators=(",", ":"))

    cxdb.add_argument("--context-id", required=True)

    return parser


def main() -> None:
    parser = build_parser()
    args = parser.parse_args()
    client = TiDBClient(TiDBConfig.from_env())
    compiler = MemoryCompiler(client, self_name=args.self_name)

    try:
        if args.command == "vault-health":
            stats = compiler.ingest_vault_health(args.vault_sqlite, limit=args.limit, container_tag=args.container_tag)
        elif args.command == "vault-twitter-bookmarks":
            stats = compiler.ingest_vault_twitter_bookmarks(args.vault_sqlite, limit=args.limit, container_tag=args.container_tag)
        elif args.command == "longmemeval":
            stats = compiler.ingest_longmemeval_style(args.jsonl, limit=args.limit, container_tag=args.container_tag)
        elif args.command == "cxdb-context":
            stats = ingest_cxdb_context(
                compiler,
                cxdb_url=args.cxdb_url,
                context_id=args.context_id,
                container_tag=args.container_tag,
            )
        else:
            raise ValueError(f"Unknown command: {args.command}")
    finally:
        client.close()

    print(stats)


if __name__ == "__main__":
    main()

    for raw_statement in sql_text.split(";"):
        statement = raw_statement.strip()
        if not statement:

from __future__ import annotations

import argparse
import shutil
from pathlib import Path


def patch_once(path: Path, old: str, new: str) -> None:
    text = path.read_text(encoding="utf-8")
    if new in text:
        return
    if old not in text:
        raise RuntimeError(f"Expected snippet not found in {path}")
    path.write_text(text.replace(old, new), encoding="utf-8")


def main() -> None:
    parser = argparse.ArgumentParser(description="Install the memo-ree provider into a MemoryBench checkout.")
    parser.add_argument("--memorybench-dir", required=True)
    args = parser.parse_args()

    repo_root = Path(__file__).resolve().parents[1]
    source_dir = repo_root / "integrations" / "memorybench" / "memoree"
    target_root = Path(args.memorybench_dir).resolve()
    target_provider_dir = target_root / "src" / "providers" / "memoree"
    target_provider_dir.mkdir(parents=True, exist_ok=True)

    for filename in ("index.ts", "prompts.ts"):
        shutil.copy2(source_dir / filename, target_provider_dir / filename)

    provider_types = target_root / "src" / "types" / "provider.ts"
    patch_once(
        provider_types,
        'export type ProviderName = "supermemory" | "mem0" | "zep" | "filesystem" | "rag"',
        'export type ProviderName = "supermemory" | "mem0" | "zep" | "filesystem" | "rag" | "memoree"',
    )

    providers_index = target_root / "src" / "providers" / "index.ts"
    patch_once(
        providers_index,
        'import { RAGProvider } from "./rag"',
        'import { RAGProvider } from "./rag"\nimport { MemoReeProvider } from "./memoree"',
    )
    patch_once(
        providers_index,
        '  rag: RAGProvider,\n}',
        '  rag: RAGProvider,\n  memoree: MemoReeProvider,\n}',
    )
    patch_once(
        providers_index,
        'export { SupermemoryProvider, Mem0Provider, ZepProvider, FilesystemProvider, RAGProvider }',
        'export { SupermemoryProvider, Mem0Provider, ZepProvider, FilesystemProvider, RAGProvider, MemoReeProvider }',
    )

    config_file = target_root / "src" / "utils" / "config.ts"
    patch_once(
        config_file,
        "  googleApiKey: string\n}",
        "  googleApiKey: string\n  memoReeApiKey: string\n  memoReeBaseUrl: string\n}",
    )
    patch_once(
        config_file,
        '  googleApiKey: process.env.GOOGLE_API_KEY || "",\n}',

    )
    patch_once(
        config_file,
        '    case "rag":\n      return { apiKey: config.openaiApiKey } // RAG provider uses OpenAI for embeddings\n',
        '    case "rag":\n      return { apiKey: config.openaiApiKey } // RAG provider uses OpenAI for embeddings\n    case "memoree":\n      return { apiKey: config.memoReeApiKey, baseUrl: config.memoReeBaseUrl }\n',
    )

    cli_index = target_root / "src" / "cli" / "index.ts"
    patch_once(
        cli_index,
        "  rag            Hybrid RAG memory (OpenClaw/QMD style)\n                 Extracts memories via LLM, chunks + embeds extracted content, hybrid BM25 + vector search.\n                 Requires: OPENAI_API_KEY (for memory extraction via gpt-4o-mini + embeddings)\n",
        "  rag            Hybrid RAG memory (OpenClaw/QMD style)\n                 Extracts memories via LLM, chunks + embeds extracted content, hybrid BM25 + vector search.\n                 Requires: OPENAI_API_KEY (for memory extraction via gpt-4o-mini + embeddings)\n\n  memoree        TiDB-backed temporal memory compiler with query-time system_memory composition\n                 Requires: MEMO_REE_BASE_URL (local API) and model keys for MemoryBench answer/judge phases\n",
    )
    patch_once(
        cli_index,
        "  -p rag            Use hybrid RAG memory (OpenClaw/QMD style)\n",
        "  -p rag            Use hybrid RAG memory (OpenClaw/QMD style)\n  -p memoree        Use memo-ree via the local benchmark API\n",
    )

    print(f"Installed memo-ree provider into {target_root}")
    print("Next steps:")
    print("  1. Start the local API: uv run python scripts/serve_benchmark_api.py --port 8787")
    print("  2. Set MEMO_REE_BASE_URL=http://127.0.0.1:8787")
    print("  3. Run MemoryBench with provider `memoree`")


if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

from memo_ree.benchmark import BenchmarkEvaluator
from memo_ree.composer import QueryTimeMemoryComposer
from memo_ree.tidb import TiDBClient, TiDBConfig


def main() -> None:
    parser = argparse.ArgumentParser(description="Compare flat observation retrieval vs temporal memory.")
    parser.add_argument("--limit", type=int, default=100)
    parser.add_argument("--self-name", default="Navan")
    parser.add_argument("--container-tag", default="default")
    args = parser.parse_args()

    client = TiDBClient(TiDBConfig.from_env())
    try:
        evaluator = BenchmarkEvaluator(client, QueryTimeMemoryComposer(client, self_name=args.self_name, default_container_tag=args.container_tag))
        results = evaluator.evaluate(limit=args.limit, container_tag=args.container_tag)
        for label, result in results.items():
            print(f"{label}: {result.correct}/{result.total} = {result.accuracy:.3f}")
            for row in result.rows:
                print(f"  {row['case_id']}: expected={row['expected']!r} answer={row['answer']!r} correct={row['correct']}")
    finally:
        client.close()
