$outputFile = "SearchResults.csv"    # Output CSV

# List of text/document extensions to include
$extensions = @("*.txt","*.csv","*.html","*.htm","*.tex","*.docx","*.xlsx","*.pdf")

# ----- SCRIPT -----
# Clear output file if it exists
if (Test-Path $outputFile) { Remove-Item $outputFile }

foreach ($ext in $extensions) {
    Get-ChildItem -Path $folder -Recurse -File -Filter $ext | ForEach-Object {
        $file = $_
        try {
            # Read content depending on file type
            if ($file.Extension -eq ".docx") {
                $word = New-Object -ComObject Word.Application
                $word.Visible = $false
                $doc = $word.Documents.Open($file.FullName)
                $content = $doc.Content.Text
                $doc.Close()
                $word.Quit()
            } elseif ($file.Extension -eq ".xlsx") {
                $excel = New-Object -ComObject Excel.Application
                $excel.Visible = $false
                $wb = $excel.Workbooks.Open($file.FullName)
                $content = ""
                foreach ($sheet in $wb.Sheets) {
                    $range = $sheet.UsedRange
                    $content += ($range.Text2 -join " ")
                }
                $wb.Close($false)
                $excel.Quit()
            } elseif ($file.Extension -eq ".pdf") {
                # PDFs: needs PDF module or library
                $content = "PDF content extraction not implemented"
            } else {
                $content = Get-Content -Path $file.FullName -Raw -ErrorAction Stop
            }

            # Stop at first match
            $found = $patterns | Where-Object { $content -cmatch $_ } | Select-Object -First 1
            if ($found) {
                [PSCustomObject]@{
                    FilePath = $file.FullName
                    CreationTime = $file.CreationTime
                    Match = $found
                } | Export-Csv $outputFile -NoTypeInformation -Append
            }

        } catch {
            # Skip files that fail to read
        }
    }
}

Write-Host "Search complete. Results saved to $outputFile"
