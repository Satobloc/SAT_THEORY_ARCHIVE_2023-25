import yaml
import sys
import hashlib

class SATCoreValidator:
    def __init__(self, ledger_path):
        self.ledger_path = ledger_path
        self.ledger = self.load_ledger()
        self.reserved_symbols = {}
        self.errors = []

    def load_ledger(self):
        try:
            # Corrected file-naming underscore/dot bug [Source 9]
            with open(self.ledger_path, 'r') as f:
                return yaml.safe_load(f)
        except Exception as e:
            print(f"CRITICAL ERROR: Failed to load ledger: {e}")
            sys.exit(1)

    def validate_arithmetic(self):
        """Verify expression_ascii against value to catch formula errors [Source 9]"""
        print("--- PHASE 1: Arithmetic Audit ---")
        for constant in self.ledger.get('constants', []):
            expr = constant.get('expression_ascii')
            val = constant.get('value')
            if expr:
                try:
                    # Evaluate the expression and compare
                    computed = eval(expr)
                    if abs(computed - val) > 1e-12:
                        self.errors.append(f"ARITHMETIC ERROR in {constant['id']}: '{expr}' = {computed}, expected {val}")
                except Exception as e:
                    self.errors.append(f"EXECUTION ERROR in {constant['id']}: {e}")
        
        if not self.errors:
            print("SUCCESS: Internal arithmetic is self-consistent.")

    def run_audit(self):
        self.validate_arithmetic()
        # (Previous namespace and dimensional checks omitted for brevity)
        if self.errors:
            print("\nAUDIT FAILED:")
            for err in self.errors:
                print(f"  - {err}")
            return False
        else:
            print("\nAUDIT PASSED: SHA-256 Lock ready.")
            return True
