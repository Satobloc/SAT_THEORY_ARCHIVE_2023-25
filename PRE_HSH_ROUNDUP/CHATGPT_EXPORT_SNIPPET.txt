(() => {
  "use strict";

  if (window.__nathanChatExporterLoaded) {
    document.getElementById("local-chatgpt-conversation-exporter-button")?.click();
    return;
  }
  window.__nathanChatExporterLoaded = true;

  const APP_ID = "local-chatgpt-conversation-exporter";
  const BUTTON_ID = `${APP_ID}-button`;
  const PANEL_ID = `${APP_ID}-panel`;

  const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

  function sanitizeFilename(value) {
    return String(value || "ChatGPT conversation")
      .replace(/[<>:"/\\|?*\u0000-\u001F]/g, "_")
      .replace(/\s+/g, " ")
      .trim()
      .slice(0, 180) || "ChatGPT conversation";
  }

  function conversationId() {
    const match = location.pathname.match(/\/c\/([^/?#]+)/);
    return match ? decodeURIComponent(match[1]) : null;
  }

  async function fetchJson(url, options = {}) {
    const response = await fetch(url, {
      credentials: "include",
      cache: "no-store",
      ...options,
      headers: {
        Accept: "application/json",
        ...(options.headers || {}),
      },
    });

    let body = null;
    try {
      body = await response.json();
    } catch {
      body = null;
    }

    return { response, body };
  }

  async function fetchConversation() {
    const id = conversationId();
    if (!id) {
      throw new Error("This does not appear to be an ordinary ChatGPT conversation URL.");
    }

    const endpoint = `/backend-api/conversation/${encodeURIComponent(id)}`;

    // In many sessions, the authenticated cookies are sufficient.
    let attempt = await fetchJson(endpoint);
    if (attempt.response.ok && attempt.body) return attempt.body;

    // Some sessions require the bearer token used by the ChatGPT web app.
    const session = await fetchJson("/api/auth/session");
    const token = session.body?.accessToken || session.body?.access_token;

    if (token) {
      attempt = await fetchJson(endpoint, {
        headers: { Authorization: `Bearer ${token}` },
      });
      if (attempt.response.ok && attempt.body) return attempt.body;
    }

    const status = attempt.response?.status || "unknown";
    throw new Error(
      `ChatGPT returned HTTP ${status}. The internal conversation endpoint may have changed, ` +
      `or this browser session may need to be refreshed.`
    );
  }

  function partToText(part) {
    if (part == null) return "";
    if (typeof part === "string") return part;

    if (Array.isArray(part)) {
      return part.map(partToText).filter(Boolean).join("\n");
    }

    if (typeof part === "object") {
      if (typeof part.text === "string") return part.text;
      if (typeof part.content === "string") return part.content;

      const pointer =
        part.asset_pointer ||
        part.image_asset_pointer ||
        part.audio_asset_pointer ||
        part.file_id;

      if (pointer) {
        const type = part.content_type || part.type || "attachment";
        return `[${type}: ${pointer}]`;
      }

      // Keep otherwise-unrecognized structured content rather than silently losing it.
      try {
        return `\n\`\`\`json\n${JSON.stringify(part, null, 2)}\n\`\`\`\n`;
      } catch {
        return String(part);
      }
    }

    return String(part);
  }

  function messageText(message) {
    const content = message?.content;
    if (!content) return "";

    if (Array.isArray(content.parts)) {
      return content.parts.map(partToText).filter(Boolean).join("\n").trim();
    }

    if (typeof content.text === "string") return content.text.trim();
    if (typeof content === "string") return content.trim();

    return partToText(content).trim();
  }

  function currentBranchNodes(conversation) {
    const mapping = conversation?.mapping;
    if (!mapping || typeof mapping !== "object") {
      throw new Error("The conversation JSON did not contain a recognizable message map.");
    }

    // Following parent pointers from current_node exports the branch actually shown in the UI,
    // rather than mixing regenerated or alternative responses together.
    const current = conversation.current_node;
    if (current && mapping[current]) {
      const nodes = [];
      const seen = new Set();
      let id = current;

      while (id && mapping[id] && !seen.has(id)) {
        seen.add(id);
        nodes.push({ id, ...mapping[id] });
        id = mapping[id].parent || null;
      }

      return nodes.reverse();
    }

    // Fallback: chronological order if current_node is unavailable.
    return Object.entries(mapping)
      .map(([id, node]) => ({ id, ...node }))
      .sort((a, b) => {
        const at = Number(a.message?.create_time || 0);
        const bt = Number(b.message?.create_time || 0);
        return at - bt;
      });
  }

  function extractMessages(conversation) {
    return currentBranchNodes(conversation)
      .map((node, index) => {
        const message = node.message;
        if (!message || typeof message !== "object") return null;

        const text = messageText(message);
        if (!text) return null;

        const role = String(message.author?.role || "unknown");
        const name = message.author?.name || null;
        const createTime = Number(message.create_time || 0);
        const updateTime = Number(message.update_time || 0);

        return {
          sequence: index + 1,
          node_id: node.id,
          role,
          author_name: name,
          created_at: createTime ? new Date(createTime * 1000).toISOString() : null,
          updated_at: updateTime ? new Date(updateTime * 1000).toISOString() : null,
          text,
          metadata: message.metadata || {},
        };
      })
      .filter(Boolean)
      .map((message, index) => ({ ...message, sequence: index + 1 }));
  }

  function localDateStart(value) {
    if (!value) return null;
    const date = new Date(`${value}T00:00:00`);
    return Number.isNaN(date.getTime()) ? null : date;
  }

  function localDateEnd(value) {
    if (!value) return null;
    const date = new Date(`${value}T23:59:59.999`);
    return Number.isNaN(date.getTime()) ? null : date;
  }

  function applyFilters(messages, settings) {
    const keyword = settings.keyword.trim().toLocaleLowerCase();
    const start = localDateStart(settings.startDate);
    const end = localDateEnd(settings.endDate);

    return messages.filter((message) => {
      if (!settings.includeInternal && !["user", "assistant"].includes(message.role)) {
        return false;
      }

      if (settings.speaker !== "all" && message.role !== settings.speaker) {
        return false;
      }

      if (keyword && !message.text.toLocaleLowerCase().includes(keyword)) {
        return false;
      }

      if (start || end) {
        if (!message.created_at) return false;
        const created = new Date(message.created_at);
        if (start && created < start) return false;
        if (end && created > end) return false;
      }

      return true;
    });
  }

  function roleLabel(message) {
    const role = message.role.toUpperCase();
    return message.author_name ? `${role} — ${message.author_name}` : role;
  }

  function renderMarkdown(title, sourceUrl, messages, settings) {
    const lines = [
      `# ${title}`,
      "",
      `Source: ${sourceUrl}`,
      "",
      `Exported: ${new Date().toISOString()}`,
      "",
      `Messages: ${messages.length}`,
      "",
      "---",
      "",
    ];

    for (const message of messages) {
      let heading = `## ${message.sequence}. ${roleLabel(message)}`;
      if (settings.includeTimestamps && message.created_at) {
        heading += ` (${message.created_at})`;
      }

      lines.push(heading, "", message.text.trimEnd(), "", "---", "");
    }

    return lines.join("\n");
  }

  function renderText(title, sourceUrl, messages, settings) {
    const lines = [
      title,
      "=".repeat(Math.max(3, title.length)),
      `Source: ${sourceUrl}`,
      `Exported: ${new Date().toISOString()}`,
      `Messages: ${messages.length}`,
      "",
    ];

    for (const message of messages) {
      let heading = `[${message.sequence}] ${roleLabel(message)}`;
      if (settings.includeTimestamps && message.created_at) {
        heading += ` — ${message.created_at}`;
      }

      lines.push(heading, "-".repeat(heading.length), message.text.trimEnd(), "");
    }

    return lines.join("\n");
  }

  function csvCell(value) {
    const text = value == null ? "" : String(value);
    return `"${text.replace(/"/g, '""')}"`;
  }

  function renderCsv(messages) {
    const rows = [
      ["sequence", "role", "author_name", "created_at", "updated_at", "text"],
      ...messages.map((message) => [
        message.sequence,
        message.role,
        message.author_name || "",
        message.created_at || "",
        message.updated_at || "",
        message.text,
      ]),
    ];

    return rows.map((row) => row.map(csvCell).join(",")).join("\n");
  }

  function splitTranscript(text, maxMegabytes) {
    const maxBytes = Math.floor(Number(maxMegabytes || 0) * 1024 * 1024);
    if (!maxBytes || new Blob([text]).size <= maxBytes) return [text];

    // Split on message dividers when possible.
    const blocks = text.split(/\n(?=## \d+\. )/);
    const parts = [];
    let current = "";

    for (const block of blocks) {
      const candidate = current ? `${current}\n${block}` : block;
      if (current && new Blob([candidate]).size > maxBytes) {
        parts.push(current);
        current = block;
      } else {
        current = candidate;
      }
    }

    if (current) parts.push(current);
    return parts;
  }

  function download(filename, content, mimeType) {
    const blob = content instanceof Blob
      ? content
      : new Blob([content], { type: mimeType });

    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = filename;
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    link.remove();

    setTimeout(() => URL.revokeObjectURL(url), 15_000);
  }

  function field(labelText, control) {
    const wrapper = document.createElement("label");
    wrapper.style.display = "grid";
    wrapper.style.gap = "4px";
    wrapper.style.fontSize = "12px";

    const label = document.createElement("span");
    label.textContent = labelText;
    label.style.fontWeight = "600";

    wrapper.append(label, control);
    return wrapper;
  }

    const input = document.createElement("input");
    input.type = type;

      color: "CanvasText",

    return input;
  }

  function styledSelect(options) {
    const select = document.createElement("select");
    for (const [value, label] of options) {
      const option = document.createElement("option");
      option.value = value;
      option.textContent = label;
      select.appendChild(option);
    }

    Object.assign(select.style, {
      width: "100%",
      boxSizing: "border-box",
      padding: "7px 8px",
      border: "1px solid #8886",
      borderRadius: "6px",
      background: "Canvas",
      color: "CanvasText",
    });

    return select;
  }

  function checkbox(labelText, checked = false) {
    const label = document.createElement("label");
    Object.assign(label.style, {
      display: "flex",
      alignItems: "center",
      gap: "7px",
      fontSize: "12px",
    });

    const input = document.createElement("input");
    input.type = "checkbox";
    input.checked = checked;

    const text = document.createElement("span");
    text.textContent = labelText;

    label.append(input, text);
    return { label, input };
  }

  function button(text, primary = false) {
    const element = document.createElement("button");
    element.type = "button";
    element.textContent = text;
    Object.assign(element.style, {
      border: primary ? "1px solid #111" : "1px solid #8886",
      borderRadius: "7px",
      padding: "8px 10px",
      cursor: "pointer",
      background: primary ? "#111" : "Canvas",
      color: primary ? "#fff" : "CanvasText",
      fontWeight: primary ? "700" : "600",
    });
    return element;
  }

  function readSettings(controls) {
    return {
      format: controls.format.value,
      speaker: controls.speaker.value,
      keyword: controls.keyword.value,
      startDate: controls.startDate.value,
      endDate: controls.endDate.value,
      maxMegabytes: controls.maxMegabytes.value,
      includeTimestamps: controls.includeTimestamps.checked,
      includeInternal: controls.includeInternal.checked,
      separateSpeakers: controls.separateSpeakers.checked,
    };
  }

  function setStatus(status, text, isError = false) {
    status.textContent = text;
    status.style.color = isError ? "#b42318" : "CanvasText";
  }

  async function performExport(controls, status, rawJson = false) {
    const settings = readSettings(controls);
    setStatus(status, "Loading structured conversation data…");

    try {
      const conversation = await fetchConversation();
      const title = sanitizeFilename(conversation.title || document.title || "ChatGPT conversation");
      const rawMessages = extractMessages(conversation);
      const filtered = applyFilters(rawMessages, settings);

      if (rawJson) {
        download(
          `${title} — raw.json`,
          JSON.stringify(conversation, null, 2),
          "application/json;charset=utf-8"
        );
        setStatus(status, `Saved raw conversation JSON (${rawMessages.length} extracted messages).`);
        return;
      }

      if (!filtered.length) {
        throw new Error("No messages matched the selected filters.");
      }

      const groups = settings.separateSpeakers
        ? [
            ["user", filtered.filter((message) => message.role === "user")],
            ["assistant", filtered.filter((message) => message.role === "assistant")],
            ["other", filtered.filter((message) => !["user", "assistant"].includes(message.role))],
          ].filter(([, group]) => group.length)
        : [["conversation", filtered]];

      let downloadCount = 0;

      for (const [groupName, messages] of groups) {
        let content;
        let extension;
        let mimeType;

        if (settings.format === "json") {
          content = JSON.stringify(
            {
              title: conversation.title || title,
              conversation_id: conversationId(),
              source_url: location.href,
              exported_at: new Date().toISOString(),
              filters: settings,
              messages,
            },
            null,
            2
          );
          extension = "json";
          mimeType = "application/json;charset=utf-8";
        } else if (settings.format === "csv") {
          content = renderCsv(messages);
          extension = "csv";
          mimeType = "text/csv;charset=utf-8";
        } else if (settings.format === "txt") {
          content = renderText(conversation.title || title, location.href, messages, settings);
          extension = "txt";
          mimeType = "text/plain;charset=utf-8";
        } else {
          content = renderMarkdown(conversation.title || title, location.href, messages, settings);
          extension = "md";
          mimeType = "text/markdown;charset=utf-8";
        }

        const parts =
          extension === "md" || extension === "txt"
            ? splitTranscript(content, settings.maxMegabytes)
            : [content];

        for (let index = 0; index < parts.length; index++) {
          const groupSuffix = settings.separateSpeakers ? ` — ${groupName}` : "";
          const partSuffix =
            parts.length > 1
              ? ` — part ${String(index + 1).padStart(3, "0")} of ${String(parts.length).padStart(3, "0")}`
              : "";

          download(`${title}${groupSuffix}${partSuffix}.${extension}`, parts[index], mimeType);
          downloadCount++;
          await sleep(250);
        }
      }

      setStatus(
        status,
        `Saved ${filtered.length} message${filtered.length === 1 ? "" : "s"} in ` +
        `${downloadCount} file${downloadCount === 1 ? "" : "s"}.`
      );
    } catch (error) {
      console.error("[ChatGPT exporter]", error);
      setStatus(status, error?.message || String(error), true);
    }
  }

  function buildPanel() {
    document.getElementById(PANEL_ID)?.remove();

    const panel = document.createElement("section");
    panel.id = PANEL_ID;
    Object.assign(panel.style, {
      position: "fixed",
      right: "18px",
      bottom: "68px",
      zIndex: "2147483647",
      width: "330px",
      maxWidth: "calc(100vw - 36px)",
      maxHeight: "calc(100vh - 100px)",
      overflow: "auto",
      boxSizing: "border-box",
      padding: "14px",
      border: "1px solid #8886",
      borderRadius: "12px",
      boxShadow: "0 12px 36px rgba(0,0,0,.25)",
      background: "Canvas",
      color: "CanvasText",
      fontFamily: "system-ui, -apple-system, sans-serif",
    });

    const title = document.createElement("div");
    title.textContent = "Export this conversation";
    title.style.fontWeight = "800";
    title.style.marginBottom = "10px";

    const format = styledSelect([
      ["md", "Markdown (.md)"],
      ["txt", "Plain text (.txt)"],
      ["json", "Filtered JSON (.json)"],
      ["csv", "CSV (.csv)"],
    ]);

    const speaker = styledSelect([
      ["all", "User + assistant"],
      ["user", "User only"],
      ["assistant", "Assistant only"],
    ]);

    const keyword = styledInput("search");
    keyword.placeholder = "Optional word or phrase";

    const startDate = styledInput("date");
    const endDate = styledInput("date");

    const maxMegabytes = styledInput("number");
    maxMegabytes.min = "0";
    maxMegabytes.step = "1";
    maxMegabytes.value = "0";
    maxMegabytes.title = "0 means one file";

    const includeTimestamps = checkbox("Include timestamps", true);
    const includeInternal = checkbox("Include system/tool messages", true);
    const separateSpeakers = checkbox("Save each speaker separately", false);

    const exportButton = button("Export transcript", true);
    const rawButton = button("Save raw JSON");
    const closeButton = button("Close");

    const status = document.createElement("div");
    Object.assign(status.style, {
      minHeight: "18px",
      marginTop: "10px",
      fontSize: "12px",
      lineHeight: "1.35",
    });
    status.textContent = "No scrolling required. Current response branch only.";

    const grid = document.createElement("div");
    Object.assign(grid.style, {
      display: "grid",
      gridTemplateColumns: "1fr 1fr",
      gap: "9px",
    });

    const fullWidth = (element) => {
      element.style.gridColumn = "1 / -1";
      return element;
    };

    grid.append(
      field("Format", format),
      field("Speakers", speaker),
      fullWidth(field("Keyword filter", keyword)),
      field("Start date", startDate),
      field("End date", endDate),
      fullWidth(field("Split size in MB (0 = one file)", maxMegabytes)),
      fullWidth(includeTimestamps.label),
      fullWidth(includeInternal.label),
      fullWidth(separateSpeakers.label)
    );

    const actions = document.createElement("div");
    Object.assign(actions.style, {
      display: "grid",
      gridTemplateColumns: "1fr 1fr",
      gap: "8px",
      marginTop: "12px",
    });

    exportButton.style.gridColumn = "1 / -1";
    actions.append(exportButton, rawButton, closeButton);

    const controls = {
      format,
      speaker,
      keyword,
      startDate,
      endDate,
      maxMegabytes,
      includeTimestamps: includeTimestamps.input,
      includeInternal: includeInternal.input,
      separateSpeakers: separateSpeakers.input,
    };

    exportButton.addEventListener("click", () => performExport(controls, status, false));
    rawButton.addEventListener("click", () => performExport(controls, status, true));
    closeButton.addEventListener("click", () => panel.remove());

    panel.append(title, grid, actions, status);
    document.body.appendChild(panel);
  }

  function addButton() {
    if (document.getElementById(BUTTON_ID) || !conversationId()) return;

    const exportButton = document.createElement("button");
    exportButton.id = BUTTON_ID;
    exportButton.type = "button";
    exportButton.textContent = "Export chat";
    exportButton.title = "Export this ChatGPT conversation";

    Object.assign(exportButton.style, {
      position: "fixed",
      right: "18px",
      bottom: "18px",
      zIndex: "2147483646",
      border: "1px solid #7776",
      borderRadius: "999px",
      padding: "9px 13px",
      cursor: "pointer",
      background: "#111",
      color: "#fff",
      font: "700 13px system-ui, -apple-system, sans-serif",
      boxShadow: "0 6px 18px rgba(0,0,0,.22)",
    });

    exportButton.addEventListener("click", buildPanel);
    document.body.appendChild(exportButton);
  }

  function watchNavigation() {
    let previousUrl = location.href;

    const observer = new MutationObserver(() => {
      if (location.href !== previousUrl) {
        previousUrl = location.href;
        document.getElementById(BUTTON_ID)?.remove();
        document.getElementById(PANEL_ID)?.remove();
      }
      addButton();
    });

    observer.observe(document.documentElement, {
      childList: true,
      subtree: true,
    });

    addButton();
  }

  watchNavigation();
  buildPanel();
})();
