-- ======================================================================
-- I. Numeric / Algebraic Mass Spectrum
-- ======================================================================

-- Bare mass scale
def m0 (T ℓ_f : ℝ) : ℝ := T / ℓ_f  -- or T*ℓ_f/c^2 if needed

-- Topological invariant Q
structure TopoInv :=
(L_wind L_link W_writhe : ℤ)

def Q (t : TopoInv) : ℤ := t.L_wind + t.L_link + t.W_writhe

-- Topological mass
def m_topo (m0 : ℝ) (t : TopoInv) : ℝ := m0 / (Q t : ℝ)

-- Induced mass
def m_ind (g1 : ℝ) (M : ℝ) (E_overlap : ℝ) : ℝ := (g1^2 / M^2) * E_overlap

-- Effective mass
def m_eff (m0 g1 M E_overlap : ℝ) (t : TopoInv) : ℝ :=
  m_topo m0 t + m_ind g1 M E_overlap

-- Quadratic excitation formula
def m2 (k β a : ℝ) (N : ℕ) (t : TopoInv) : ℝ :=
  k * (N + β * ((Q t)^2 : ℝ) - a)
