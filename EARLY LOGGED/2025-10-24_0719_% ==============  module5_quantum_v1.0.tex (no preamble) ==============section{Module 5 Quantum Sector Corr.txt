\section{Module 5: Quantum Sector Corrections}

\paragraph{Version.} v1.0 (20 Jun 2025)

\subsection{Scope and Validation Targets}
This module folds one- and two-loop quantum corrections into the gauge,
flavour, and $\Sigma_t$ sectors while preserving the no–fine–tune rule.
Success criteria:
\begin{enumerate}
  \item Compute oblique electroweak parameters $(S,T,U)$ with \emph{no} ad-hoc counterterms,
        matching the PDG 2025 global fit within $1\,\sigma$.
  \item Demonstrate vacuum stability of the $\Sigma_t$ potential with lifetime
        $\tau_{\mathrm{vac}}>10^{100}$ y.
  \item Show that all divergences are absorbed by filament wave–function
        renormalisation factors $Z_F$ and $Z_\Sigma$
        that are analytic in $\T\Lf^{2}$.
\end{enumerate}

\subsection{WP5.1 Gauge‑Vertex Renormalisation}
One-loop correction to the filament three-link vertex:
\[
\delta\Gamma^{(1)}=-\frac{g_G^3}{16\pi^2}\,b_G\,
\ln\!\frac{\mu^2}{\T/\Lf},
\qquad
b_G = \Bigl\{\tfrac{41}{6},\ -\tfrac{19}{6},\ -7\Bigr\}
_{G=U(1),SU(2),SU(3)}.
\]
Absorbed by $Z_F=1+\delta Z_F$ with
$\delta Z_F = -\delta\Gamma^{(1)}/g_G$.

\subsection{WP5.3 Vacuum Stability of $\Sigma_t$}

\[
V_{\rm eff}(\Sigma_t)=
\tfrac12 m_\Sigma^2\Sigma_t^2
+\tfrac{\lambda}{4}\Sigma_t^4
+\Delta V^{(2)}(\Sigma_t),
\]

$\lambda(\mu) > 0$ up to $\mu=10^{3}M_*$.
