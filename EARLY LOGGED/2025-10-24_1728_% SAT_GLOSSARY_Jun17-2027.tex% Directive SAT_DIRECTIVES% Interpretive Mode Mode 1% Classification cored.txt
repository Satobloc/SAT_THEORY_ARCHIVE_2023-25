\section*{Glossary of Symbols and Terms}

\begin{description}
  \item[$\theta_4(x)$] Scalar field representing the local angular deviation between filament tangent vector $v^\mu$ and the emergent time-flow vector $u^\mu(x)$. Diagnostic only; not a source of mass.
  \item[$u^\mu(x)$] Emergent time-flow vector field derived from local filament current density. Normalized as $u^\mu u_\mu = -1$.
  \item[$S_{\mu\nu}(x)$] Strain tensor encoding local bundle distortion, defined as:
  \[
    S_{\mu\nu} = \nabla_\mu u_\nu + \nabla_\nu u_\mu.
  \]
  \item[$S(x)$] Strain scalar, defined as $S(x) = \sqrt{S_{\mu\nu}(x)\,S^{\mu\nu}(x)}$. Quantifies local misalignment of the time vector field and governs emergent proper time rate and entropy production.
  \item[$\rho_{\rm wind}(x), \rho_{\rm link}(x), \rho_{\rm triplet}(x)$] Topological density fields counting filament winding, linking, and triple-linking respectively. Used to define gauge coupling constants.
  \item[$g_G(x)$] Emergent gauge coupling constant, defined via
  \[
    g_G^{-2}(x) \sim \rho_G(x)\,\ell_f^2,
  \]
  where $G\in\{\mathrm{U}(1),\mathrm{SU}(2),\mathrm{SU}(3)\}$.
  \item[$Q$] Topological suppression index combining winding, linking, and writhing of filament bundles. Appears in
  \[
    m_{\rm eff} = \frac{m_0}{Q}.
  \]
  \item[$\phi(x)$] Foliation scalar field labeling hypersurfaces of constant emergent time. Gradient yields $u^\mu$.
  \item[$d\tau(x)$] Emergent proper time increment, defined locally as $d\tau = S(x)\,d\phi$.
\end{description}
